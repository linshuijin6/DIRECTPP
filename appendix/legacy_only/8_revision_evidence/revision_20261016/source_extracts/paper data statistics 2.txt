import numpy as np
from scipy.stats import f_oneway, chi2_contingency

# 数据初始化
np.random.seed(42)  # 设置随机种子以保证结果可重复

# 各组样本数量
n_AD, n_CN, n_EMCI, n_LMCI, n_MCI = 317, 577, 291, 152, 520

# 年龄 (Age)
age_AD = np.random.normal(74.90, 7.99, n_AD)
age_CN = np.random.normal(72.55, 6.65, n_CN)
age_EMCI = np.random.normal(71.62, 7.51, n_EMCI)
age_LMCI = np.random.normal(72.44, 7.69, n_LMCI)
age_MCI = np.random.normal(73.20, 7.51, n_MCI)

# 体重 (Weight)
weight_AD = np.random.normal(73.21, 18.64, n_AD)
weight_CN = np.random.normal(73.20, 24.13, n_CN)
weight_EMCI = np.random.normal(80.77, 16.67, n_EMCI)
weight_LMCI = np.random.normal(77.15, 16.64, n_LMCI)
weight_MCI = np.random.normal(72.90, 22.33, n_MCI)

# MMSE
mmse_AD = np.random.normal(20.84, 1.97, n_AD)
mmse_CN = np.random.normal(29.40, 0.70, n_CN)
mmse_EMCI = np.random.normal(27.50, 0.50, n_EMCI)
mmse_LMCI = np.random.normal(24.47, 0.50, n_LMCI)
mmse_MCI = np.random.normal(26.83, 1.24, n_MCI)

# GDSCALE
gdscale_AD = np.random.normal(4.14, 1.62, n_AD)
gdscale_CN = np.random.normal(0.24, 0.73, n_CN)
gdscale_EMCI = np.random.normal(1.01, 0.14, n_EMCI)
gdscale_LMCI = np.random.normal(3.02, 0.28, n_LMCI)
gdscale_MCI = np.random.normal(1.81, 0.99, n_MCI)

# Global CDR
global_cdr_AD = np.random.normal(0.95, 0.15, n_AD)
global_cdr_CN = np.random.normal(0.00, 0.02, n_CN)
global_cdr_EMCI = np.random.normal(0.15, 0.23, n_EMCI)
global_cdr_LMCI = np.random.normal(0.50, 0.01, n_LMCI)
global_cdr_MCI = np.random.normal(0.42, 0.19, n_MCI)

# FAQ
faq_AD = np.random.normal(20.20, 2.92, n_AD)
faq_CN = np.random.normal(0.52, 0.53, n_CN)
faq_EMCI = np.random.normal(3.11, 2.42, n_EMCI)
faq_LMCI = np.random.normal(8.83, 1.54, n_LMCI)
faq_MCI = np.random.normal(5.33, 1.51, n_MCI)

# NPI-Q
npiq_AD = np.random.normal(4.98, 0.65, n_AD)
npiq_CN = np.random.normal(0.01, 0.19, n_CN)
npiq_EMCI = np.random.normal(1.55, 2.17, n_EMCI)
npiq_LMCI = np.random.normal(4.00, 0.03, n_LMCI)
npiq_MCI = np.random.normal(2.92, 0.50, n_MCI)

# 性别比例 (Sex(M/F))
sex_AD = [186, 131]
sex_CN = [252, 325]
sex_EMCI = [165, 126]
sex_LMCI = [80, 72]
sex_MCI = [310, 210]

# 构造性别数据表
sex_table = np.array([sex_AD, sex_CN, sex_EMCI, sex_LMCI, sex_MCI])

# 计算 P 值
p_age = f_oneway(age_AD, age_CN, age_EMCI, age_LMCI, age_MCI).pvalue
p_weight = f_oneway(weight_AD, weight_CN, weight_EMCI, weight_LMCI, weight_MCI).pvalue
p_sex = chi2_contingency(sex_table)[1]  # 卡方检验
p_mmse = f_oneway(mmse_AD, mmse_CN, mmse_EMCI, mmse_LMCI, mmse_MCI).pvalue

p_faq = f_oneway(faq_AD, faq_CN, faq_EMCI, faq_LMCI, faq_MCI).pvalue
p_npiq = f_oneway(npiq_AD, npiq_CN, npiq_EMCI, npiq_LMCI, npiq_MCI).pvalue

# 输出结果
print(f"P(Age): {p_age:.6f}")
print(f"P(Weight): {p_weight:.6f}")
print(f"P(Sex): {p_sex:.6f}")
